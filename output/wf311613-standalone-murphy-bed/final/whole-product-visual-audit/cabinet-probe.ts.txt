import * as THREE from 'three';import {GeometryFactory} from '../../../../src/engine/geometry/GeometryFactory';import {MaterialFactory} from '../../../../src/engine/materials/MaterialFactory';
const scene=new THREE.Scene();scene.background=new THREE.Color('#bbb7af');const materials=Object.fromEntries(Object.entries(window.probeData.materials).map(([id,def])=>[id,MaterialFactory.create(def)]));const meshes=[];for(const part of window.probeData.items){if(!part.geometry)continue;const geometry=GeometryFactory.create(part.geometry),def=window.probeData.materials[part.material];if(def?.finish==='oak-review'){const pos=geometry.getAttribute('position'),n=geometry.getAttribute('normal'),uv=new Float32Array(pos.count*2),axis=def.grainAxis??'x';for(let i=0;i<pos.count;i++){const long=axis==='x'?pos.getX(i):axis==='y'?pos.getY(i):pos.getZ(i);const transverse=axis==='x'?(Math.abs(n.getY(i))>.5?pos.getZ(i):pos.getY(i)):axis==='z'?(Math.abs(n.getY(i))>.5?pos.getX(i):pos.getY(i)):pos.getX(i);uv[i*2]=long/90;uv[i*2+1]=transverse/24;}geometry.setAttribute('uv',new THREE.BufferAttribute(uv,2));}const mesh=new THREE.Mesh(geometry,materials[part.material]??materials['oak-x']);mesh.userData.id=part.id;mesh.userData.category=part.category;mesh.matrixAutoUpdate=false;mesh.matrix.fromArray(part.world);mesh.castShadow=mesh.receiveShadow=true;scene.add(mesh);meshes.push(mesh);}scene.add(new THREE.AmbientLight(0xffffff,.32));scene.add(new THREE.HemisphereLight(0xffffff,0xbbb9b5,.65));const key=new THREE.DirectionalLight(0xfffaf2,2.2);key.position.set(-70,160,-110);key.castShadow=true;key.shadow.mapSize.set(2048,2048);Object.assign(key.shadow.camera,{left:-150,right:150,top:100,bottom:-100,near:.1,far:450});key.shadow.normalBias=.025;key.shadow.camera.updateProjectionMatrix();scene.add(key);const fill=new THREE.DirectionalLight(0xedf3ff,.9);fill.position.set(80,100,120);scene.add(fill);
const camera=new THREE.PerspectiveCamera(35,1280/720,.1,5000);camera.position.set(59,205,-130);camera.lookAt(new THREE.Vector3(59,193.5,18));const renderer=new THREE.WebGLRenderer({antialias:true,preserveDrawingBuffer:true});renderer.setSize(1280,720);renderer.shadowMap.enabled=true;renderer.shadowMap.type=THREE.PCFSoftShadowMap;document.body.style.margin='0';document.body.appendChild(renderer.domElement);
window.probe=(mode)=>{renderer.shadowMap.enabled=mode!=='no-shadow';key.castShadow=mode!=='no-shadow';for(const material of Object.values(materials))material.needsUpdate=true;for(const mesh of meshes)mesh.castShadow=mode==='wood-shadow'?mesh.userData.category!=='hardware':mode==='hardware-shadow'?mesh.userData.category==='hardware':true;renderer.render(scene,camera);};window.probe('shadow');
