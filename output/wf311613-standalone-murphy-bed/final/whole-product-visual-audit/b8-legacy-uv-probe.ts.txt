import * as THREE from 'three';import {GeometryFactory} from '../../../../src/engine/geometry/GeometryFactory';import {MaterialFactory} from '../../../../src/engine/materials/MaterialFactory';
const scene=new THREE.Scene();scene.background=new THREE.Color('#bbb7af');const g=GeometryFactory.create({type:'box',size:[113.6,49,1.2],bevel:.04}),m=MaterialFactory.create({type:'standard',color:'#947857',finish:'oak-review',grainAxis:'x',roughness:.55});const pos=g.getAttribute('position'),normal=g.getAttribute('normal'),uv=new Float32Array(pos.count*2);for(let i=0;i<pos.count;i++){uv[i*2]=pos.getX(i)/90;uv[i*2+1]=(Math.abs(normal.getY(i))>.5?pos.getZ(i):pos.getY(i))/24;}g.setAttribute('uv',new THREE.BufferAttribute(uv,2));const mesh=new THREE.Mesh(g,m);mesh.position.set(-59,137.5,18);mesh.castShadow=mesh.receiveShadow=true;scene.add(mesh);scene.add(new THREE.AmbientLight(0xffffff,.32));scene.add(new THREE.HemisphereLight(0xffffff,0xbbb9b5,.65));const key=new THREE.DirectionalLight(0xfffaf2,2.2);key.position.set(-70,160,-110);key.castShadow=true;key.shadow.mapSize.set(2048,2048);Object.assign(key.shadow.camera,{left:-150,right:150,top:100,bottom:-100,near:.1,far:450});key.shadow.normalBias=.025;scene.add(key);const fill=new THREE.DirectionalLight(0xedf3ff,.9);fill.position.set(80,100,120);scene.add(fill);
const camera=new THREE.PerspectiveCamera(35,1280/720,.1,5000);camera.position.set(-59,145,-160);camera.lookAt(mesh.position);const renderer=new THREE.WebGLRenderer({antialias:true,preserveDrawingBuffer:true});renderer.setSize(1280,720);renderer.shadowMap.enabled=true;renderer.shadowMap.type=THREE.PCFSoftShadowMap;document.body.style.margin='0';document.body.appendChild(renderer.domElement);
window.probe=(mode)=>{mesh.material=mode==='flat-map'?new THREE.MeshBasicMaterial({map:m.map}):mode==='normals'?new THREE.MeshNormalMaterial():m;renderer.shadowMap.enabled=mode==='shadow';renderer.render(scene,camera);};window.probe('shadow');
